function fmt(n) {

function today() { return toISO(new Date()); }
function toISO(d) {
  const y = d.getFullYear();
  const m = String(d.getMonth() + 1).padStart(2, '0');
  const dd = String(d.getDate()).padStart(2, '0');
  return `${y}-${m}-${dd}`;
}
function addDays(dateStr, n) {
  const d = new Date(dateStr + 'T00:00:00');
  d.setDate(d.getDate() + n);
  return toISO(d);
}
function getWeekStart(dateStr, weekStartDay) {
  const d = new Date(dateStr + 'T00:00:00');
  const day = d.getDay();
  let diff = day - weekStartDay;
  if (diff < 0) diff += 7;
  d.setDate(d.getDate() - diff);
  return toISO(d);
}
function getWeekNumber(dateStr) {
  const d = new Date(dateStr + 'T00:00:00');
  const target = new Date(d.valueOf());
  const dayNr = (d.getDay() + 6) % 7;
  target.setDate(target.getDate() - dayNr + 3);
  const firstThursday = target.valueOf();
  target.setMonth(0, 1);
  if (target.getDay() !== 4) {
    target.setMonth(0, 1 + ((4 - target.getDay()) + 7) % 7);
  }
  return 1 + Math.ceil((firstThursday - target) / 604800000);
}

  } finally {
    if (showOverlay) hideLoading();
  }
}

async function createRecord(rec) {
  const { data, error } = await supabase
    .from('records')
    .insert([{
      date: rec.date,
      type: rec.type,
      category: rec.category,
      amount: rec.amount,
      note: rec.note || ''
    }])
    .select()
    .single();
  if (error) throw error;
  return {
    id: data.id,
    date: data.date,
    type: data.type,
    category: data.category,
    amount: Number(data.amount),
    note: data.note || ''
  };
}

async function updateRecordById(id, updates) {

      note: updates.note || ''

    .select()

  return {
    id: data.id,
    date: data.date,
    type: data.type,
    category: data.category,
    amount: Number(data.amount),
    note: data.note || ''
  };
}

function calcTotals() {
  let income = 0, expense = 0;
  for (const r of state.records) {
    if (r.type === '收入') income += r.amount;
    else expense += r.amount;
  }
  return { income: round2(income), expense: round2(expense), balance: round2(income - expense) };
}

function getCurrentWeekRange() {
  const start = getWeekStart(today(), state.weekStartDay);
  return { start, end: addDays(start, 6) };
}
function getWeekStats(range) {
  let income = 0, expense = 0, count = 0;
  for (const r of state.records) {
    if (r.date < range.start || r.date > range.end) continue;
    count++;
    if (r.type === '收入') income += r.amount;
    else expense += r.amount;
  }
  return { income: round2(income), expense: round2(expense), count };
}

function buildWeeklyReport() {
  const wsDay = state.weekStartDay;
  const map = new Map();
  for (const r of state.records) {
    const ws = getWeekStart(r.date, wsDay);
    if (!map.has(ws)) map.set(ws, { income: 0, expense: 0, count: 0 });
    const g = map.get(ws);
    g.count++;
    if (r.type === '收入') g.income += r.amount;
    else g.expense += r.amount;
  }
  const weeks = [...map.keys()].sort();
  let running = 0;
  const report = [];
  for (const ws of weeks) {
    const g = map.get(ws);
    const net = round2(g.income - g.expense);
    running = round2(running + net);
    report.push({
      weekStart: ws,
      weekEnd: addDays(ws, 6),
      weekNumber: getWeekNumber(ws),
      income: round2(g.income),
      expense: round2(g.expense),
      net,
      cumulative: running,
      count: g.count
    });
  }
  return report;
}

function matchPieRange(r) {
  if (pieRange === 'all') return true;
  const range = getCurrentWeekRange();
  if (pieRange === 'week') return r.date >= range.start && r.date <= range.end;
  if (pieRange === 'month') {
    const now = new Date();
    const prefix = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`;
    return r.date.startsWith(prefix);
  }
  return true;
}

function aggregateByCategory(records) {
  const agg = new Map();
  let total = 0;
  for (const r of records) {
    agg.set(r.category, (agg.get(r.category) || 0) + r.amount);
    total += r.amount;
  }
  const items = [...agg.entries()]
    .map(([name, value]) => ({ name, value: round2(value) }))
    .sort((a, b) => b.value - a.value);
  return { items, total: round2(total) };
}

function drawRingSegment(ctx, cx, cy, rOuter, rInner, startAngle, endAngle, color) {
  ctx.beginPath();
  ctx.arc(cx, cy, rOuter, startAngle, endAngle, false);
  ctx.arc(cx, cy, rInner, endAngle, startAngle, true);
  ctx.closePath();
  ctx.fillStyle = color;
  ctx.fill();
  ctx.strokeStyle = '#fff';
  ctx.lineWidth = 2;
  ctx.stroke();
}

function drawExpenseRatioPie(incomeData, expenseData) {
  const canvas = document.getElementById('pie-canvas');
  const dpr = window.devicePixelRatio || 1;
  const cssSize = Math.max(canvas.clientWidth || 380, 240);
  canvas.width = Math.round(cssSize * dpr);
  canvas.height = Math.round(cssSize * dpr);
  const ctx = canvas.getContext('2d');
  ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
  ctx.clearRect(0, 0, cssSize, cssSize);

  const cx = cssSize / 2;
  const cy = cssSize / 2;
  const radius = cssSize / 2 - 4;
  const innerR = radius * 0.55;

  const incomeTotal = incomeData.total;
  const expenseTotal = expenseData.total;
  const hasIncome = incomeTotal > 0;
  const ratio = hasIncome ? expenseTotal / incomeTotal : 0;
  const scale = (hasIncome && ratio > 1) ? 1 / ratio : 1;

  drawRingSegment(ctx, cx, cy, radius, innerR, 0, Math.PI * 2, '#e2e8f0');

  let startAngle = -Math.PI / 2;
  if (expenseTotal > 0 && expenseData.items.length > 0) {
    expenseData.items.forEach((item, i) => {
      const portion = hasIncome ? item.value / incomeTotal : item.value / expenseTotal;
      const slice = portion * Math.PI * 2 * scale;
      const color = PIE_COLORS_EXPENSE[i % PIE_COLORS_EXPENSE.length];
      item.color = color;
      drawRingSegment(ctx, cx, cy, radius, innerR, startAngle, startAngle + slice, color);
      startAngle += slice;
    });
  }

  ctx.textAlign = 'center';
  ctx.textBaseline = 'middle';

  if (!hasIncome) {
    ctx.fillStyle = '#64748b';
    ctx.font = '600 12px "Microsoft JhengHei UI", "PingFang TC", sans-serif';
    ctx.fillText('此範圍無收入', cx, cy - 10);
    ctx.fillStyle = '#1e293b';
    ctx.font = '700 18px "Microsoft JhengHei UI", "PingFang TC", sans-serif';
    ctx.fillText('—', cx, cy + 14);
  } else {
    const pct = ratio * 100;
    const pctStr = pct.toFixed(1) + '%';
    let pctColor;
    if (pct > 100) pctColor = '#991b1b';
    else if (pct > 80) pctColor = '#c2410c';
    else if (pct > 50) pctColor = '#d97706';
    else pctColor = '#16a34a';

    ctx.fillStyle = '#64748b';
    ctx.font = '600 11px "Microsoft JhengHei UI", "PingFang TC", sans-serif';
    ctx.fillText('支出占收入', cx, cy - innerR * 0.55);

    const bigSize = Math.max(16, Math.min(26, innerR * 0.34));
    ctx.fillStyle = pctColor;
    ctx.font = `800 ${bigSize}px "Microsoft JhengHei UI", "PingFang TC", sans-serif`;
    ctx.fillText(pctStr, cx, cy + innerR * 0.02);

    const subSize = Math.max(9, Math.min(11, innerR * 0.14));
    ctx.fillStyle = '#64748b';
    ctx.font = `600 ${subSize}px "Microsoft JhengHei UI", "PingFang TC", sans-serif`;
    const subText = `$${fmt(expenseTotal)} / $${fmt(incomeTotal)}`;
    if (ctx.measureText(subText).width > innerR * 1.6) {
      ctx.font = `600 ${subSize - 1}px "Microsoft JhengHei UI", "PingFang TC", sans-serif`;
    }
    ctx.fillText(subText, cx, cy + innerR * 0.55);
  }

  renderLegend(incomeData, expenseData, ratio);
}

function renderLegend(incomeData, expenseData, ratio) {
  const legendEl   = document.getElementById('pie-expense-legend');
  const ratioBarEl = document.getElementById('ratio-bar');

  const incomeTotal  = incomeData.total;
  const expenseTotal = expenseData.total;
  const balance      = round2(incomeTotal - expenseTotal);
  const hasIncome    = incomeTotal > 0;

  document.getElementById('pie-income-total').textContent  = '$' + fmt(incomeTotal);
  document.getElementById('pie-expense-total').textContent = '$' + fmt(expenseTotal);
  const balEl = document.getElementById('pie-balance-total');
  balEl.textContent = (balance < 0 ? '-$' : '$') + fmt(Math.abs(balance));
  balEl.style.color = balance >= 0 ? 'var(--success)' : 'var(--danger)';

  if (expenseData.items.length === 0) {
    legendEl.innerHTML = '<div class="pie-section-empty">此範圍內無支出記錄</div>';
    document.getElementById('pie-expense-ratio').textContent = '$0.00';
  } else {
    document.getElementById('pie-expense-ratio').textContent =
      hasIncome ? (ratio * 100).toFixed(1) + '%' : '$' + fmt(expenseTotal);

    let html = expenseData.items.map(item => {
      const pct = hasIncome
        ? (item.value / incomeTotal * 100).toFixed(1) + '%'
        : (item.value / expenseTotal * 100).toFixed(1) + '%';
      return `
        <div class="pie-legend-item">
          <span class="dot" style="background:${item.color};"></span>
          <span class="name">${esc(item.name)}</span>
          <span class="value">$${fmt(item.value)}</span>
          <span class="pct">${pct}</span>
        </div>`;
    }).join('');

    if (hasIncome && balance > 0) {
      const pct = (balance / incomeTotal * 100).toFixed(1) + '%';
      html += `
        <div class="pie-legend-item balance-item">
          <span class="dot" style="background:#e2e8f0;"></span>
          <span class="name" style="color:var(--muted);">結餘（未花費）</span>
          <span class="value" style="color:var(--success);">$${fmt(balance)}</span>
          <span class="pct">${pct}</span>
        </div>`;
    }
    legendEl.innerHTML = html;
  }

  if (incomeTotal > 0) {
    const pct = (ratio * 100).toFixed(1);
    ratioBarEl.style.display = 'flex';
    if (ratio > 1) {
      ratioBarEl.className = 'ratio-bar over';
      ratioBarEl.innerHTML = `⚠️ 支出超過收入 <strong>${pct}%</strong>｜超支 <strong>$${fmt(Math.abs(balance))}</strong>`;
    } else if (ratio > 0.8) {
      ratioBarEl.className = 'ratio-bar';
      ratioBarEl.innerHTML = `支出佔收入 <strong>${pct}%</strong>｜剩餘 $${fmt(balance)}（${(100 - ratio * 100).toFixed(1)}%）`;
    } else {
      ratioBarEl.className = 'ratio-bar ok';
      ratioBarEl.innerHTML = `支出佔收入 <strong>${pct}%</strong>｜結餘 $${fmt(balance)}（${(100 - ratio * 100).toFixed(1)}%）`;
    }
  } else if (expenseTotal > 0) {
    ratioBarEl.style.display = 'flex';
    ratioBarEl.className = 'ratio-bar over';
    ratioBarEl.innerHTML = `此範圍內無收入記錄，無法計算支出佔比`;
  } else {
    ratioBarEl.style.display = 'none';
  }
}

function renderPie() {
  const labels = { week: '本週', month: '本月', all: '全部' };
  document.getElementById('pie-range-label').textContent = `｜${labels[pieRange]}`;

  const incomeRecords  = state.records.filter(r => r.type === '收入' && matchPieRange(r));
  const expenseRecords = state.records.filter(r => r.type === '支出' && matchPieRange(r));

  const incomeData  = aggregateByCategory(incomeRecords);
  const expenseData = aggregateByCategory(expenseRecords);

  drawExpenseRatioPie(incomeData, expenseData);
}

function render() {

  document.getElementById('s-total-income').textContent = '$' + fmt(totals.income);
  document.getElementById('s-total-expense').textContent = '$' + fmt(totals.expense);

  balEl.style.color = totals.balance >= 0 ? 'var(--success)' : 'var(--danger)';

  const range = getCurrentWeekRange();
  const weekStats = getWeekStats(range);
  const weekNet = round2(weekStats.income - weekStats.expense);

  document.getElementById('week-start-select').value = String(state.weekStartDay);
  document.getElementById('current-week-label').textContent = `｜第 ${getWeekNumber(range.start)} 週`;
  document.getElementById('s-week-range').textContent = `${range.start} ～ ${range.end}`;
  document.getElementById('s-week-income').textContent = '$' + fmt(weekStats.income);
  document.getElementById('s-week-expense').textContent = '$' + fmt(weekStats.expense);
  const wnEl = document.getElementById('s-week-net-2');
  wnEl.textContent = (weekNet >= 0 ? '+' : '-') + '$' + fmt(Math.abs(weekNet));
  wnEl.style.color = weekNet >= 0 ? 'var(--success)' : 'var(--danger)';

  const wn2El = document.getElementById('s-week-net');
  wn2El.textContent = (weekNet >= 0 ? '+' : '-') + '$' + fmt(Math.abs(weekNet));
  wn2El.style.color = weekNet >= 0 ? 'var(--success)' : 'var(--danger)';

  const fMonth = document.getElementById('flt-month').value;
  const fType = document.getElementById('flt-type').value;
  const fKw = document.getElementById('flt-keyword').value.trim().toLowerCase();

  let rows = state.records.slice();
  if (fMonth) rows = rows.filter(r => r.date.startsWith(fMonth));
  if (fType) rows = rows.filter(r => r.type === fType);
  if (fKw) rows = rows.filter(r =>
    (r.note || '').toLowerCase().includes(fKw) ||
    (r.category || '').toLowerCase().includes(fKw)
  );
  rows.sort((a, b) => (b.date.localeCompare(a.date)) || (String(b.id).localeCompare(String(a.id))));

  const tbody = document.getElementById('tbody');
  const emptyMsg = document.getElementById('empty-msg');

  if (rows.length === 0) {
    tbody.innerHTML = '';
    emptyMsg.style.display = 'block';
    emptyMsg.textContent = state.records.length === 0
      ? '尚無記錄，從上方新增第一筆吧！'
      : '沒有符合條件的記錄';
  } else {
    emptyMsg.style.display = 'none';
    tbody.innerHTML = rows.map(r => {
      const cls = r.type === '收入' ? 'income' : 'expense';
      return `
        <tr class="${cls}">
          <td class="date">${esc(r.date)}</td>
          <td><span class="badge ${cls}">${esc(r.type)}</span></td>
          <td>${esc(r.category)}</td>
          <td class="amount">${r.type === '收入' ? '+' : '-'}$${fmt(r.amount)}</td>
          <td>${esc(r.note || '')}</td>
          <td style="text-align:center; white-space:nowrap;">
            <button class="btn-ghost btn-sm" onclick="openEdit('${String(r.id)}')">編輯</button>
            <button class="btn-danger btn-sm" onclick="del('${String(r.id)}')">刪除</button>
          </td>
        </tr>`;
    }).join('');
  }
  document.getElementById('summary').textContent = `共 ${rows.length} 筆記錄`;

  const weekTbody = document.getElementById('week-tbody');
  const weekEmpty = document.getElementById('week-empty');
  const report = buildWeeklyReport();

  if (report.length === 0) {
    weekTbody.innerHTML = '';
    weekEmpty.style.display = 'block';
    weekEmpty.textContent = '尚無週結算資料';
  } else {
    weekEmpty.style.display = 'none';
    const currentWeekStart = range.start;
    weekTbody.innerHTML = report.slice().reverse().map(w => {
      const isCurrent = w.weekStart === currentWeekStart;
      const netCls = w.net >= 0 ? 'var(--success)' : 'var(--danger)';
      const cumCls = w.cumulative >= 0 ? 'var(--success)' : 'var(--danger)';
      return `
        <tr class="week-row ${isCurrent ? 'current' : ''}">
          <td>
            <strong>第 ${w.weekNumber} 週</strong>
            ${isCurrent ? '<span class="badge current">本週</span>' : ''}
          </td>
          <td class="date">${w.weekStart} ～ ${w.weekEnd}</td>
          <td class="amount" style="color:var(--success);">+$${fmt(w.income)}</td>
          <td class="amount" style="color:var(--danger);">-$${fmt(w.expense)}</td>
          <td class="amount" style="color:${netCls};font-weight:700;">${w.net >= 0 ? '+' : '-'}$${fmt(Math.abs(w.net))}</td>
          <td class="amount" style="color:${cumCls};">$${fmt(w.cumulative)}</td>
          <td class="amount" style="color:var(--muted);">${w.count}</td>
        </tr>`;
    }).join('');
  }

  // 資料管理面板
  document.getElementById('cloud-count').textContent = state.records.length + ' 筆';
  if (lastSyncTime) {
    const hh = String(lastSyncTime.getHours()).padStart(2, '0');
    const mm = String(lastSyncTime.getMinutes()).padStart(2, '0');
    const ss = String(lastSyncTime.getSeconds()).padStart(2, '0');
    document.getElementById('last-sync').textContent = `${hh}:${mm}:${ss}`;
  }

  renderPie();
}

function exportCSV() {
  if (state.records.length === 0) return toast('沒有記錄可匯出', 'error');
  const rows = state.records.slice().sort((a, b) => a.date.localeCompare(b.date));
  const lines = [['日期', '週次', '類型', '分類', '金額', '備註'].join(',')];
  for (const r of rows) {
    const wk = getWeekNumber(getWeekStart(r.date, state.weekStartDay));
    lines.push([
      r.date, `第${wk}週`, r.type, r.category, r.amount.toFixed(2),
      '"' + String(r.note || '').replace(/"/g, '""') + '"'
    ].join(','));
  }
  downloadFile('\ufeff' + lines.join('\n'), `記帳明細_${today()}.csv`, 'text/csv;charset=utf-8');
}

function exportWeeklyCSV() {
  const report = buildWeeklyReport();
  if (report.length === 0) return toast('沒有週結算資料可匯出', 'error');
  const lines = [['週次', '起始日', '結束日', '本週收入', '本週支出', '本週淨額', '累計結餘', '筆數'].join(',')];
  for (const w of report) {
    lines.push([
      `第${w.weekNumber}週`, w.weekStart, w.weekEnd,
      w.income.toFixed(2), w.expense.toFixed(2),
      w.net.toFixed(2), w.cumulative.toFixed(2), w.count
    ].join(','));
  }
  downloadFile('\ufeff' + lines.join('\n'), `記帳週報_${today()}.csv`, 'text/csv;charset=utf-8');
}

function downloadFile(content, filename, mime) {
  const blob = new Blob([content], { type: mime });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  a.remove();
  URL.revokeObjectURL(url);
}

/* ============================================================
   匯入 JSON（批次上傳到 Supabase）
   ============================================================ */
async function importJSON(evt) {
  const file = evt.target.files[0];
  if (!file) return;
  if (!supabase) { toast('尚未設定 Supabase', 'error'); evt.target.value = ''; return; }

  const reader = new FileReader();
  reader.onload = async (e) => {
    try {
      const data = JSON.parse(e.target.result);
      const recs = Array.isArray(data) ? data : data.records;
      if (!Array.isArray(recs)) throw new Error('檔案格式不正確（找不到 records 陣列）');
      if (recs.length === 0) throw new Error('檔案中沒有任何記錄');

      if (!confirm(`確定要匯入 ${recs.length} 筆記錄到雲端嗎？\n\n這會「追加」到現有資料中，不會刪除原有記錄。`)) {
        evt.target.value = '';
        return;
      }

      showLoading();
      let success = 0, failed = 0;
      for (const rec of recs) {
        try {
          const amount = round2(Number(rec.amount));
          if (!Number.isFinite(amount) || amount <= 0) throw new Error('金額錯誤');
          if (!rec.date || !/^\d{4}-\d{2}-\d{2}$/.test(rec.date)) throw new Error('日期錯誤');
          if (rec.type !== '收入' && rec.type !== '支出') throw new Error('類型錯誤');

          await supabase.from('records').insert([{
            date: rec.date,
            type: rec.type,
            category: rec.category || '其他',
            amount,
            note: rec.note || ''
          }]);
          success++;
        } catch (err) {
          failed++;
        }
      }
      hideLoading();
      await loadData(false);
      toast(`匯入完成：成功 ${success} 筆${failed ? `，失敗 ${failed} 筆` : ''}`, failed ? 'error' : 'success');
    } catch (err) {
      hideLoading();
      toast('匯入失敗：' + err.message, 'error');
    }
    evt.target.value = '';
  };
  reader.readAsText(file);
}

/* ============================================================
   清除所有資料
   ============================================================ */
async function clearAllData() {
  if (!supabase) return;
  if (state.records.length === 0) return toast('目前沒有資料可清除', 'error');

  if (!confirm(`⚠️ 確定要清除雲端所有 ${state.records.length} 筆資料嗎？\n\n此操作無法復原。建議先「下載備份」。`)) return;
  if (!confirm('再次確認：真的要清空整個帳本嗎？')) return;

  showLoading();
  try {
    const { error } = await supabase
      .from('records')
      .delete()
      .neq('id', 0);  // 刪除所有 id ≠ 0 的記錄（即全部）
    if (error) throw error;
    state.records = [];
    render();
    toast('已清除雲端所有資料', 'success');
    setSyncStatus('ok', '已同步');
  } catch (err) {
    toast('清除失敗：' + err.message, 'error');
  } finally {
    hideLoading();
  }
}
